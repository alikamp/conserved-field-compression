def load(name, shape=(100,500,500)):
    p = sorted(glob.glob(f'**/{name}*.f32', recursive=True))[0]
    return np.ascontiguousarray(np.fromfile(p, dtype='<f4').reshape(shape))
print("fields present:", len(glob.glob('**/*f48*.f32', recursive=True)))

# --- Cell 2: ZFP sweep on real data, 3D + 2D slice + 1D line-out ------------
# (real coded ratios; this is E1 for ZFP + the dimensionality axis E3)
from rd_full import study, print_rows, ZFP, lineout
f3 = load('TCf48')                       # 3D temperature
rows  = study(f3,          "ISABEL-TC-3d", ZFP())
rows += study(f3[50],      "ISABEL-TC-2d", ZFP())     # a 2D slice
rows += study(lineout(f3), "ISABEL-TC-1d", ZFP())     # 1D line-out (scientific 1D series)
print_rows(rows)
import csv
with open('rd_zfp.csv','w',newline='') as fp:
    w=csv.DictWriter(fp, fieldnames=rows[0].keys()); w.writeheader(); w.writerows(rows)
print("saved rd_zfp.csv")
