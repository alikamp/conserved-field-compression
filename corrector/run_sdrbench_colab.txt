%%writefile rd_conserve.py
import sys, numpy as np, zfpy

def synth_field(n=64, seed=0):
    rng = np.random.default_rng(seed)
    fk = np.zeros((n,n,n), complex); K = 6
    fk[:K,:K,:K] = rng.standard_normal((K,K,K)) + 1j*rng.standard_normal((K,K,K))
    bg = np.fft.ifftn(fk).real; bg = bg/bg.std()
    xs = np.linspace(-1,1,n); X,Y,Z = np.meshgrid(xs,xs,xs, indexing='ij')
    f = bg.copy()
    for _ in range(6):
        cx,cy,cz = rng.uniform(-0.6,0.6,3); w = rng.uniform(0.05,0.15); amp = rng.uniform(1,4)
        f += amp*np.exp(-((X-cx)**2+(Y-cy)**2+(Z-cz)**2)/(2*w**2))
    return f.astype(np.float32)

if len(sys.argv) > 1:
    f0 = np.load(sys.argv[1]).astype(np.float32); src = sys.argv[1]
else:
    f0 = synth_field(64, 0); src = "synthetic coherent field"

n = f0.size
M0 = float(np.sum(f0.astype(np.float64)))
E0 = float(np.sum((f0.astype(np.float64))**2))
V0 = float(np.var(f0.astype(np.float64)))
rng_span = float(f0.max() - f0.min())
print(f"field: {src}   shape={f0.shape}  {n} values  range={rng_span:.4g}\n")

def psnr(a, b):
    mse = np.mean((a.astype(np.float64)-b.astype(np.float64))**2)
    return np.inf if mse == 0 else 20*np.log10(rng_span) - 10*np.log10(mse)
def invariants(f):
    fd = f.astype(np.float64); return float(np.sum(fd)), float(np.sum(fd**2))
def conserve(f_hat):
    fd = f_hat.astype(np.float64); mu = fd.mean(); v = fd.var()
    a = np.sqrt(V0/v) if v > 0 else 1.0
    fc = mu + a*(fd - mu)
    fc = fc + (M0 - fc.sum())/n
    return fc.astype(np.float32), a

hdr = (f"{'tol/range':>10}{'ratio':>7}{'PSNR':>7}"
       f"{'maxE/tol raw':>12}{'maxE/tol cor':>13}"
       f"{'E drift raw':>12}{'E drift cor':>12}"
       f"{'M drift raw':>12}{'M drift cor':>12}")
print(hdr); print('-'*len(hdr))
for frac in [1e-2, 3e-3, 1e-3, 3e-4, 1e-4, 3e-5]:
    tol = frac * rng_span
    comp = zfpy.compress_numpy(f0, tolerance=tol)
    fh   = zfpy.decompress_numpy(comp)
    ratio= f0.nbytes/len(comp)
    maxerr = float(np.abs(f0.astype(np.float64)-fh.astype(np.float64)).max())
    Mh, Eh = invariants(fh)
    Edr_raw = abs(Eh-E0)/abs(E0); Mdr_raw = abs(Mh-M0)/abs(M0)
    fc, a = conserve(fh)
    Mc, Ec = invariants(fc)
    Edr_c = abs(Ec-E0)/abs(E0); Mdr_c = abs(Mc-M0)/abs(M0)
    maxerr_c = float(np.abs(f0.astype(np.float64)-fc.astype(np.float64)).max())
    print(f"{frac:>10.0e}{ratio:>7.1f}{psnr(f0,fh):>7.1f}"
          f"{maxerr/tol:>12.2f}{maxerr_c/tol:>13.2f}"
          f"{Edr_raw:>12.2e}{Edr_c:>12.2e}"
          f"{Mdr_raw:>12.2e}{Mdr_c:>12.2e}")
print(f"\nlayer cost: 8 bytes  (adds {8*8/n:.2e} bits/value)")
print("maxE/tol<1 => ZFP bound held; identical raw vs cor => layer preserves it.")
print("E/M drift 'raw' grows with compression; 'cor' stays ~machine for 8 bytes.")

import glob, numpy as np
# CLOUD is sparse (mostly zero); a DENSER field tells a cleaner conservation
# story -- prefer temperature (TC), pressure (P), or velocity (U/V/W).
cand = sorted(glob.glob('**/TC*f48*.f32', recursive=True)
              or glob.glob('**/Uf48*.f32', recursive=True)
              or glob.glob('**/*.f32', recursive=True))
print("using:", cand[0])
f = np.fromfile(cand[0], dtype='<f4').reshape(100, 500, 500)   # little-endian float32
np.save('field.npy', f)
print("shape", f.shape, "range", float(f.max()-f.min()))
